cmake_minimum_required(VERSION 3.16)
project(ai_box_loitering_detect VERSION 0.1.0 LANGUAGES C CXX)
option(AIBOX_WITH_AWNN "Build the vendor AWNN adapter (requires the real board SDK)" OFF)
option(AIBOX_BUILD_TESTS "Build hardware-independent rule tests" ON)
set(AIBOX_AWNN_SDK_ROOT "" CACHE PATH "Allwinner ai-sdk checkout containing examples/yolov5 and viplite-tina")
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp)
if(NOT NLOHMANN_JSON_INCLUDE_DIR)
  message(FATAL_ERROR "Install nlohmann-json3-dev (nlohmann/json.hpp is required)")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0 gstreamer-video-1.0 gstreamer-rtsp-1.0)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(aibox_rules src/behavior/engines.cpp)
target_include_directories(aibox_rules PUBLIC include)
target_compile_options(aibox_rules PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_config src/config.cpp)
target_include_directories(aibox_config PUBLIC include PRIVATE ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(aibox_config PUBLIC aibox_rules)
target_compile_options(aibox_config PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_event_store src/storage/event_store.cpp)
target_include_directories(aibox_event_store PUBLIC include)
target_link_libraries(aibox_event_store PUBLIC SQLite::SQLite3 Threads::Threads)
target_compile_options(aibox_event_store PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_snapshot src/storage/snapshot_writer.cpp)
target_include_directories(aibox_snapshot PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aibox_snapshot PUBLIC aibox_event_store ${OpenCV_LIBS} Threads::Threads)
target_compile_options(aibox_snapshot PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_tracker src/tracker/byte_tracker.cpp)
target_include_directories(aibox_tracker PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aibox_tracker PUBLIC ${OpenCV_LIBS})
target_compile_options(aibox_tracker PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_behavior src/behavior/coordinator.cpp)
target_include_directories(aibox_behavior PUBLIC include)
target_link_libraries(aibox_behavior PUBLIC aibox_rules aibox_tracker)
target_compile_options(aibox_behavior PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_yolo src/detector/yolo.cpp)
target_include_directories(aibox_yolo PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aibox_yolo PUBLIC ${OpenCV_LIBS})
target_compile_options(aibox_yolo PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_video src/video/gst_rtsp_source.cpp)
target_include_directories(aibox_video PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aibox_video PUBLIC ${OpenCV_LIBS} PkgConfig::GSTREAMER)
target_compile_options(aibox_video PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_render src/render/gst_display.cpp)
target_include_directories(aibox_render PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aibox_render PUBLIC aibox_rules ${OpenCV_LIBS} PkgConfig::GSTREAMER)
target_compile_options(aibox_render PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_preview src/render/preview_publisher.cpp)
target_include_directories(aibox_preview PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aibox_preview PUBLIC aibox_rules ${OpenCV_LIBS} Threads::Threads)
target_compile_options(aibox_preview PRIVATE -Wall -Wextra -Wpedantic)
add_library(aibox_core_api src/core_api.cpp)
target_include_directories(aibox_core_api PUBLIC include PRIVATE ${NLOHMANN_JSON_INCLUDE_DIR})
target_link_libraries(aibox_core_api PUBLIC aibox_config aibox_event_store aibox_preview Threads::Threads)
target_compile_options(aibox_core_api PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ai-box-core src/main.cpp)
target_link_libraries(ai-box-core PRIVATE aibox_rules aibox_tracker aibox_behavior aibox_config aibox_event_store aibox_snapshot aibox_video aibox_render aibox_preview aibox_core_api)
if(AIBOX_WITH_AWNN)
  if(NOT AIBOX_AWNN_SDK_ROOT)
    message(FATAL_ERROR "Set AIBOX_AWNN_SDK_ROOT to the Allwinner ai-sdk checkout")
  endif()
  set(AWNN_EXAMPLE_DIR "${AIBOX_AWNN_SDK_ROOT}/examples")
  set(VIP_DIR "${AIBOX_AWNN_SDK_ROOT}/viplite-tina/lib/aarch64-none-linux-gnu/v2.0")
  foreach(required "${AWNN_EXAMPLE_DIR}/libawnn_viplite/awnn_lib.c" "${AWNN_EXAMPLE_DIR}/libawnn_viplite/awnn_quantize.c" "${VIP_DIR}/inc/vip_lite.h" "${VIP_DIR}/libNBGlinker.so" "${VIP_DIR}/libVIPhal.so")
    if(NOT EXISTS "${required}")
      message(FATAL_ERROR "Required SDK file missing: ${required}")
    endif()
  endforeach()
  add_library(aibox_awnn_runtime STATIC
    "${AWNN_EXAMPLE_DIR}/libawnn_viplite/awnn_lib.c"
    "${AWNN_EXAMPLE_DIR}/libawnn_viplite/awnn_quantize.c")
  target_include_directories(aibox_awnn_runtime PRIVATE
    "${AWNN_EXAMPLE_DIR}/libawnn_viplite" "${VIP_DIR}/inc")
  find_library(VIP_NBGLINKER NAMES NBGlinker PATHS "${VIP_DIR}" NO_DEFAULT_PATH REQUIRED)
  find_library(VIP_HAL NAMES VIPhal PATHS "${VIP_DIR}" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(aibox_awnn_runtime PUBLIC ${VIP_NBGLINKER} ${VIP_HAL} pthread)
  add_library(aibox_awnn src/detector/awnn_detector.cpp)
  target_include_directories(aibox_awnn PUBLIC include PRIVATE "${AWNN_EXAMPLE_DIR}/libawnn_viplite" "${VIP_DIR}/inc")
  target_link_libraries(aibox_awnn PUBLIC aibox_yolo aibox_awnn_runtime)
  target_link_libraries(ai-box-core PRIVATE aibox_awnn)
  target_compile_definitions(ai-box-core PRIVATE AIBOX_WITH_AWNN=1 AIBOX_DEFAULT_MODEL_PATH="${AWNN_EXAMPLE_DIR}/yolov5/model/v3/yolov5.nb")
  add_executable(aibox-image-regression tools/run_image.cpp)
  target_link_libraries(aibox-image-regression PRIVATE aibox_awnn)
endif()
